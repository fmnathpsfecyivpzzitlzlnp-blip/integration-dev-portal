3. Агрегация и коллекции (Магия Groovy)

Методы, которые заменяют десятки строк циклов for и if.

def trades = [
    [id: 1, symbol: 'XAUUSD', profit: 150.5, type: 'BUY',  lots: 0.5, tags: ['news', 'scalp']],
    [id: 2, symbol: 'EURUSD', profit: -45.0, type: 'SELL', lots: 0.1, tags: ['grid']],
    [id: 3, symbol: 'XAUUSD', profit: 210.0, type: 'SELL', lots: 1.0, tags: ['swing']],
    [id: 4, symbol: 'GBPUSD', profit: 30.0,  type: 'BUY',  lots: 0.2, tags: ['scalp']],
    [id: 5, symbol: 'XAUUSD', profit: -15.0, type: 'BUY',  lots: 0.5, tags: ['grid']]
]

// ==========================================
// ЧАСТЬ 1: БАЗОВЫЕ МЕТОДЫ
// ==========================================

// 1. collect (Извлечь одно поле)
def symbols = trades.collect { it.symbol }.unique()
// Результат: ['XAUUSD', 'EURUSD', 'GBPUSD']

// 2. findAll (Простая фильтрация)
def profitableTrades = trades.findAll { it.profit > 0 }

// 3. sum (Быстрая сумма)
def totalProfit = trades.sum { it.profit }

// 4. groupBy (Простая группировка)
def tradesBySymbol = trades.groupBy { it.symbol }

// 5. count (Подсчет элементов)
def xauCount = trades.count { it.symbol == 'XAUUSD' }


// ==========================================
// ЧАСТЬ 2: ПРОДВИНУТЫЕ МЕТОДЫ И ПОДГРУППЫ
// ==========================================

// --- 2.1 ТРАНСФОРМАЦИЯ И ИЗВЛЕЧЕНИЕ ---
// collectEntries: Конвертация List в Map [id: profit] для быстрого поиска
def profitMap = trades.collectEntries { [ (it.id): it.profit ] }

// collectMany (flatten): Достать все теги из всех сделок в один плоский список
def allTags = trades.collectMany { it.tags }.unique()


// --- 2.2 СЛОЖНАЯ ФИЛЬТРАЦИЯ ---
// Несколько условий сразу (Золото, в плюсе, лот >= 0.5)
def heavyGoldWins = trades.findAll { it.symbol == 'XAUUSD' && it.profit > 0 && it.lots >= 0.5 }

// any / every: Проверка условий (возвращают true/false)
def hasBigLoss = trades.any { it.profit < -100 } // false
def isValidLots = trades.every { it.lots > 0 }   // true


// --- 2.3 АГРЕГАЦИЯ ---
// sum с условием: Суммарный объем только по сделкам SELL
def totalSellLots = trades.findAll { it.type == 'SELL' }.sum { it.lots }

// inject: Сложный накопительный итог (Баланс $100k минус $2 комиссии за КАЖДУЮ сделку)
def startBalance = 100000.0
def finalBalance = trades.inject(startBalance) { balance, trade ->
    balance + trade.profit - 2.0
}


// --- 2.4 ГРУППИРОВКА И СОРТИРОВКА ---
// Группировка + Вычисление: Считаем чистый PnL по каждой паре
def pnlBySymbol = trades.groupBy { it.symbol }.collectEntries { symbol, symbolTrades ->
    [ (symbol): symbolTrades.sum { it.profit } ]
}

// Сложная сортировка: Сначала по символу (A-Z), затем по профиту (по убыванию)
def sortedTrades = trades.sort { a, b ->
    a.symbol <=> b.symbol ?: b.profit <=> a.profit
}

6. HTTP POST запрос (отправка JSON)

import groovy.json.JsonBuilder

def url = new URL("http://api.local/v1/data")
def connection = url.openConnection()

// Подготавливаем JSON
def payload = new JsonBuilder([status: "ok", count: 42]).toString()

connection.setRequestMethod("POST")
connection.setDoOutput(true)
connection.setRequestProperty("Content-Type", "application/json")

// Отправляем данные
connection.outputStream.withWriter("UTF-8") { writer ->
    writer.write(payload)
}

// Читаем ответ
if (connection.responseCode in 200..299) {
    println "Успех: " + connection.inputStream.text
} else {
    println "Ошибка ${connection.responseCode}: " + connection.errorStream?.text
}

matcher.each { ip ->
    println "- $ip"
}

groupMatcher.each { fullMatch, serverName, ip ->
    println "Упал сервер: $serverName (Адрес: $ip)"
}

9. Отправка и парсинг SOAP (curl + XmlSlurper)

Идеально для интеграционных тестов. Отправляет запрос через curl и вытаскивает только нужный блок ответа (RS или Fault).

import groovy.xml.XmlSlurper
import groovy.xml.XmlUtil // Обязательно добавляем для красивого вывода структуры

def sendAndParseKpi(String soapAction, String xmlRequest) {
    def cmd = [
        "curl", "-s", "-k", "-X", "POST",
        "-u", "your_username:your_password",
        "-H", "Content-Type: text/xml; charset=utf-8",
        "-H", "SOAPAction: \"${soapAction}\"",
        "-d", "@-",
        "https://demo-6.newtontest.psb.local/api/external/soap/?service=kpi_sync"
    ]

    try {
        def process = cmd.execute()

        def os = process.getOutputStream()
        os.write(xmlRequest.getBytes("UTF-8"))
        os.flush()
        os.close()

        process.waitFor()

        def exitCode = process.exitValue()
        if (exitCode != 0) {
            return "Ошибка сети. Код завершения curl: ${exitCode}. Ошибка: ${process.err.text}"
        }

        def responseXml = process.in.text
        if (!responseXml?.trim()) {
            return "Получен пустой ответ от сервера."
        }

        /* * ВАРИАНТ 1: Если тебе нужен ВЕСЬ сырой ответ (вместе с <soap:Envelope>)
         * просто раскомментируй строку ниже:
         * return responseXml
         */

        // ВАРИАНТ 2: Парсим и возвращаем только нужную структуру (RS / Result / Fault)
        def parsed = new XmlSlurper().parseText(responseXml)

        // Ищем узел ответа (подстраивается под Response или Result)
        def rsNode = parsed.'**'.find { it.name().contains('Response') || it.name() == 'Result' }

        if (rsNode) {
            // Сериализуем найденный узел обратно в отформатированную XML-строку
            return XmlUtil.serialize(rsNode)
        } else {
            // Если успешного ответа нет, ищем ошибку SOAP Fault
            def faultNode = parsed.'**'.find { it.name() == 'Fault' }
            if (faultNode) {
                return XmlUtil.serialize(faultNode)
            }
            // Если вообще ничего не понятно, отдаем как есть
            return "Неизвестный формат ответа:\n${responseXml}"
        }

    } catch (Exception e) {
        return "Системная ошибка выполнения скрипта: ${e.message}"
    }
}
